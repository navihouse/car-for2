cmake_minimum_required(VERSION 3.22)
project(mission_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
set(MISSION_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../Core/Src/Mission)
set(MODULE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../Core/Src/Modules)
set(MOTION ${MISSION_DIR}/motion_control.c ${MISSION_DIR}/motion_distance.c
    ${MISSION_DIR}/motion_edge.c ${MISSION_DIR}/motion_straight.c

             ${MISSION_DIR}/segment_bd.c ${MISSION_DIR}/segment_da.c)
function(mission_test target)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE ${MISSION_DIR} ${MODULE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/../Core/Inc ${CMAKE_CURRENT_SOURCE_DIR})
    if(NOT MSVC)
        target_link_libraries(${target} PRIVATE m)
    endif()
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endfunction()
