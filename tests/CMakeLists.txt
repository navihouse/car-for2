cmake_minimum_required(VERSION 3.22)
project(mission_host_tests C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
set(MISSION_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../Core/Src/Mission)
set(MODULE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../Core/Src/Modules)
set(SEGMENTS ${MISSION_DIR}/segment_ac.c ${MISSION_DIR}/segment_cb.c
             ${MISSION_DIR}/segment_bd.c ${MISSION_DIR}/segment_da.c)
function(mission_test target)
    add_executable(${target} ${ARGN})
    target_include_directories(${target} PRIVATE ${MISSION_DIR} ${MODULE_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
    if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endfunction()
mission_test(mission_control_test test_mission_control.c ${MISSION_DIR}/mission_control.c ${SEGMENTS})
set(RUNTIME ${MISSION_DIR}/mission.c ${MISSION_DIR}/mission_control.c
    ${MODULE_DIR}/button.c ${MODULE_DIR}/speed_control.c mission_fake_port.c)
foreach(mode RANGE 0 4)
    mission_test(mission_placeholder_${mode} test_mission_placeholder.c ${RUNTIME} ${SEGMENTS})
    target_compile_definitions(mission_placeholder_${mode} PRIVATE MISSION_BUILD_MODE=${mode})
endforeach()
mission_test(mission_runtime_test test_mission_runtime.c mission_fake_segments.c ${RUNTIME})
